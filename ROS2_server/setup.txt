# 1. 建立工作目錄

cd ~/tm2_ws

# 2. 下載 tm2_ros2 原始碼（humble 分支）
git clone https://github.com/TechmanRobotInc/tm2_ros2.git -b humble

# 3. 修改資料夾名稱為 src（colcon 預設來源資料夾）
mv tm2_ros2 src	

# 4. 載入 ROS 2 環境（請確認路徑正確）
source ~/ros2_humble/install/local_setup.bash
# 路徑改jazzy環境
source /opt/ros/jazzy/setup.bash


# 5. 安裝缺少的相依套件（推坑紀錄）
sudo apt update
sudo apt install qtdeclarative5-dev

# 6. 編譯專案
colcon build --continue-on-error

# 7. 編譯完成後，載入工作區環境
source install/local_setup.bash
